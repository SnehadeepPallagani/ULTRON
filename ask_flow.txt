INTERNET_PROVIDERS = {
    "duckduckgo": DuckDuckGoSearchProvider()
}

ACTIVE_SEARCH_PROVIDER = "duckduckgo"


def get_internet_status():
    provider = INTERNET_PROVIDERS.get(ACTIVE_SEARCH_PROVIDER)

    return {
        "online": True,
        "layer": "ULTRON Internet Layer",
        "active_search_provider": provider.name if provider else None,
        "available_search_providers": [
            p.name for p in INTERNET_PROVIDERS.values()
        ]
    }


def internet_search(query, max_results=5):
    if not query:
        return {
            "success": False,
            "error": "Search query is empty.",
            "internet_layer": True
        }

    provider = INTERNET_PROVIDERS.get(ACTIVE_SEARCH_PROVIDER)

    if provider is None:
        return {
            "success": False,
            "query": query,
            "internet_layer": True,
            "error": "No active Internet search provider."
        }

    try:
        result = provider.search(query, max_results)
        result["internet_layer"] = True
        return result

    except Exception as e:
        return {
            "success": False,
            "query": query,
            "internet_layer": True,
            "provider": provider.name,
            "error": str(e)
        }


def web_search(query, max_results=5):
    return internet_search(query, max_results)

def _has_any_pattern(text, patterns):
    return any(re.search(pattern, text, re.IGNORECASE) for pattern in patterns)

def classify_task(prompt):
    lower=(prompt or "").strip().lower()
    if _has_any_pattern(lower, IDENTITY_PATTERNS): intent="identity"
    elif _has_any_pattern(lower, TECHNICAL_PATTERNS): intent="technical"
    elif _has_any_pattern(lower, RESEARCH_PATTERNS): intent="research"
    elif _has_any_pattern(lower, CREATIVE_PATTERNS): intent="creative"
    elif re.search(r"\b(compare|difference|versus|vs\.)\b", lower): intent="comparison"
    elif re.search(r"\b(plan|planning|roadmap|strategy)\b", lower): intent="planning"
    elif re.search(r"\b(solve|calculate|prove|derive)\b", lower): intent="problem_solving"
    elif re.search(r"\b(summarize|summary|summarise|shorten)\b", lower): intent="summarization"
    else: intent="general"
    current=_has_any_pattern(lower,CURRENT_PATTERNS); explicit=_has_any_pattern(lower,RESEARCH_PATTERNS); high=_has_any_pattern(lower,HIGH_ACCURACY_PATTERNS)
    broad=bool(re.search(r"\b(latest|major developments|trends|state of|overview|landscape)\b",lower))
    score=(5 if current else 0)+(4 if explicit else 0)+(3 if high else 0)+(2 if broad else 0)
    signals=[]
    if current: signals.append("current_or_time_sensitive")
    if explicit: signals.append("research_or_verification")
    if high: signals.append("high_accuracy_domain")
    if broad: signals.append("broad_research_topic")
    if _has_any_pattern(lower,STATIC_PATTERNS) and score<5: score-=2; signals.append("stable_knowledge_candidate")
    required=current or explicit or high or score>=5
    depth="critical" if explicit and high else "deep" if high or (explicit and broad) else "quick" if required else "none"
    return {"intent":intent,"research_required":required,"research_depth":depth,"research_score":score,"signals":signals,"high_accuracy":high,"broad":broad,"max_results":{"critical":8,"deep":7,"quick":5,"none":0}[depth]}

def build_research_queries(prompt,task):
    year=datetime.now().year; base=prompt.strip(); lower=base.lower(); queries=[base]
    if task["research_depth"] in ("deep","critical"):
        if task["high_accuracy"]:
            queries += [f"{base} authoritative sources evidence {year}",f"{base} official guidance {year}"]
        elif task["broad"] or any(x in lower for x in ("latest","developments","trends","landscape")):
            queries += [f"latest {base} {year} official announcements",f"latest {base} {year} research news"]
        elif any(x in lower for x in ("research","paper","study","literature")):
            queries += [f"{base} research papers {year}",f"{base} systematic review evidence"]
        else: queries.append(f"{base} official sources {year}")
    elif task["research_required"] and any(x in lower for x in ("latest","current","recent","news")):
        queries.append(f"{base} {year} official")
    seen=set(); out=[]
    for q in queries:
        k=q.lower().strip()
        if k and k not in seen: seen.add(k); out.append(q)
    return out[:4]

SOURCE_DOMAIN_WEIGHTS={"gov":100,"gov.in":100,"who.int":100,"nih.gov":100,"cdc.gov":100,"fda.gov":100,"nasa.gov":100,"python.org":100,"docs.python.org":100,"arxiv.org":95,"nature.com":95,"science.org":95,"acm.org":90,"ieee.org":90,"sciencedirect.com":88,"springer.com":88,"openai.com":95,"anthropic.com":95,"deepmind.google":95,"microsoft.com":92,"google.com":90,"meta.com":90,"nvidia.com":90,"reuters.com":85,"apnews.com":85,"bbc.com":82}

def source_quality(url,title=""):
    try: host=(urlparse(url).hostname or "").lower().replace("www.","")
    except Exception: host=""
    score=50
    for domain,weight in SOURCE_DOMAIN_WEIGHTS.items():
        if host==domain or host.endswith("."+domain): score=max(score,weight)
    if host.endswith(".gov") or ".gov." in host: score=max(score,100)
    if host.endswith(".edu"): score=max(score,82)
    if any(x in host for x in ("reddit.com","quora.com","facebook.com","pinterest.com")): score=min(score,35)
    return score

def rank_search_results(results):
    items=[]; seen=set()
    for result in results:
        if not result.get("success"): continue
        for raw in result.get("results",[]):
            item=dict(raw); url=item.get("url","").strip(); title=item.get("title","").strip(); key=(url or title).lower()
            if not key or key in seen: continue
            seen.add(key); item["source_quality"]=source_quality(url,title); items.append(item)
    items.sort(key=lambda x:x.get("source_quality",50),reverse=True)
    return items

def format_research_context(results,max_items=14):
    ranked=rank_search_results(results)[:max_items]
    if not ranked: return "No usable web sources were retrieved."
    return "\n\n".join(f"SOURCE {i}\nTITLE: {x.get('title','Untitled')}\nURL: {x.get('url','')}\nQUALITY SCORE: {x.get('source_quality',50)}\nEVIDENCE SNIPPET: {x.get('snippet','')}" for i,x in enumerate(ranked,1))

def clean_model_answer(answer):
    answer=(answer or "").strip().replace("\\*\\*","**").replace("\\*","*").replace("\\`","`")
    answer=re.sub(r"\[([^\]]+)\]\(\[([^\]]+)\]\((https?://[^\)]+)\)\)",r"[\1](\3)",answer)
    return answer.strip()
